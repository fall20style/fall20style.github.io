* Org mode로 jekyll page 만들기

** init.el 설정하기

#+begin_src elisp
;; org mode publishing
;; https://thackl.github.io/blogging-with-emacs-org-mode-and-jekyll
(setq org-publish-project-alist
      '(("fall20style.github.io" ;; my blog project (just a name)
         ;; Path to org files.
         :base-directory "C:/Users/fall2/work/gitdir/fall20style.github.io/_org"
         :base-extension "org"
         ;; Path to Jekyll Posts
         :publishing-directory "C:/Users/fall2/work/gitdir/fall20style.github.io/_posts/"
         :recursive t
         :publishing-function org-html-publish-to-html
         :headline-levels 4
         :html-extension "html"
         :body-only t
         )))
#+end_src
